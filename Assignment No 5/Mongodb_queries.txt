    city: "Nashik"
})

        name: "Priya",

    },

db.students.find()

db.students.findOne({ name: "Rahul" })

db.students.find({ marks: { $gt: 80 } })

    { name: "Rahul" },
    { $set: { marks: 90 } }
)

    { course: "BCA" },
    { $inc: { marks: 5 } }
)

db.students.deleteOne({ name: "Amit" })

db.students.deleteMany({ marks: { $lt: 70 } })
